/*SubConsulta com a cláusula FROM 
(Como calcular a média dos preços dos livros e retornar os livros que estão 
com preços acima da média) AVG com função de agregação
*/

SELECT L.NomeLivro, L.PrecoLivro 
FROM Livro L,
	(SELECT AVG(PrecoLivro) AS MediaPreco FROM Livro) 
	AS SubConsulta
	WHERE L.PrecoLivro > SubConsulta.MediaPreco;

/*Aninhamento de subconsultas, pode ser mais simples que usar vários INNER JOINS.
