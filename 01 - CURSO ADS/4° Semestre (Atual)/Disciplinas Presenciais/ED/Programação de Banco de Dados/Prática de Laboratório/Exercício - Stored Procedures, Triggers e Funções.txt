--Para passar dois parâmetros para o sp realizar um cálculo usando dados da tabela

ALTER PROCEDURE sp_LivroValor (
@ID as smallint, @Qtde SMALLINT)
AS
BEGIN
	SELECT NomeLivro Título, @Qtde Quantidade,
	PrecoLivro * @Qtde 'Valor Total'
	FROM Livro
	WHERE IdLivro = @ID;
END
GO

AS
BEGIN

	FROM Autor

END
GO

/* Funções UDF

Podemos criar funções (User-Defined Functions / UDFs) para realizar diversas 
tarefas no BD tais como:
- Inserir lógica complexa em uma consulta
- Criar novas funções para expressões complexas
- Substituir views com a vantagem de aceitar parâmetros

As funções UDF podem ser Escalares ou Tabulares (com valor de tabela):
- Funçã UDF Escalar: Retorna um valor único
- Funçã UDF Tabular: Retorna uma tabela de valores (DataSET)
*/

--Funções Escalares - Sintaxe

RETURNS tipo_dado_retorno
AS
BEGIN
	bloco de códigos
	RETURN valor_retorno
END;

CREATE FUNCTION valor_total(@inferior SMALLINT, @superior SMALLINT)
RETURNS REAL
AS
BEGIN
	DECLARE @media REAL
	SELECT @media = AVG(PrecoLivro)
	FROM Livro
	WHERE IdLivro BETWEEN @inferior AND @superior
	RETURN @media
END;

SELECT dbo.valor_total (101,105) Total;


--Função com valor de tabela com os dados de livros e editoras

CREATE FUNCTION retorna_itens(@valor REAL)
RETURNS TABLE
AS
RETURN (
	SELECT L.NomeLivro, A.NomeAssunto, E.NomeEditora
	FROM Livro L
	JOIN Assunto A
	ON L.IdAssunto = A.IdAssunto
	JOIN Editora E
	ON L.IdEditora = E.IdEditora
	WHERE L.PrecoLivro > @valor
);

--Aqui menciona-se os dados que se deseja ver.

SELECT NomeLivro, NomeAssunto, NomeEditora From dbo.retorna_itens (130);

SELECT NomeLivro FROM dbo.retorna_itens (62.00);

--Para excluir uma função, pocedemos da mesma maneira que fizemos com os Triggers e SPs.
