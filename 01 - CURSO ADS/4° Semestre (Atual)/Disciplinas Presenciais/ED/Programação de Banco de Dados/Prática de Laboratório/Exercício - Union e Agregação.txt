/*

Metacaracteres
% - Qualquer cadeia de 0 ou mais caracteres
_ - Qualquer caracter único
[ ] Caracteres únicos no intervalo ou conjunto especificado
[^] Caracreres único NÂO no intervalo ou conjunto especificado

*/

SELECT * FROM Livro WHERE NomeLivro LIKE 'F%';

SELECT * FROM Livro WHERE NomeLivro LIKE '[FD]%';

SELECT * FROM Livro WHERE NomeLivro LIKE '[^F^D]%';

SELECT * FROM Livro WHERE NomeLivro LIKE '[aeiou]%';

SELECT * FROM Livro WHERE NomeLivro LIKE '%[aeiou]';

SELECT * FROM Livro WHERE NomeLivro LIKE '[0-9]%';

SELECT NomeLivro FROM Livro WHERE NomeLivro LIKE 'A%o';

SELECT NomeAutor, SobrenomeAutor FROM Autor
WHERE NomeAutor LIKE 'R%' AND SobrenomeAutor LIKE 'B%';

--Lista as editora com nomes que começam com a letra O até a Letra T.

SELECT NomeEditora FROM Editora WHERE NomeEditora LIKE '[o-t]%'

--Nomes dos autores que tenha 4 letras e comecem com a letra J.
--Menciona-se a letra J e 3 sublinhados ou underscore (J___)

SELECT NomeAutor, SobrenomeAutor FROM Autor
WHERE NomeAutor LIKE 'J___' ;

/*

Livros que comecem com uma consoante e nenhum número. 
Neste caso aparecem os números, e precisamos listar apenas 
os que comecem com uma consoante.
Basta inserir no final da sequência de vogais o caracter 0 e o 9
como segue ‘[aeiou0-9]%’

*/

SELECT * FROM Livro WHERE NomeLivro NOT LIKE '[aeiou0-9]%';
