--Para excluir um bd, é necessário desconectá-lo da instância
--e também de todas as conexões existentes.
USE master;
GO
ALTER DATABASE BIBLIOTECA1
SET single_user WITH ROLLBACK immediate;
GO
DROP DATABASE BIBLIOTECA1;
GO

RESTORE DATABASE BIBLIOTECA1
FROM DISK = 'C:\SQL\BIBLIOTECA.BAK';
GO
